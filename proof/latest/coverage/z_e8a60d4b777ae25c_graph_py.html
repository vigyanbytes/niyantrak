1"""Read side: derive the current graph from all events. Pure Python; DuckDB views are an optional extra. 

2 

3Rules (status is derived, never stored): 

4- links: the latest event per (kind, src, dst) wins — add after retract makes a link live again. 

5- a decision is *superseded* when it has a live SUPERSEDED_BY link; *retired* when its latest RETIRED event is an 

6 add or its latest status is "dropped"; *pending* when its latest status is "proposed" or "to do"; otherwise 

7 *in force* (a "done" status closes a pending decision). 

8- work items take the bucket of their latest STATUS event, "to do" when none. 

9- "Latest" means the greatest (ts, write order); equal timestamps fall back to the order events were written. 

10""" 

11import json 

12import os 

13 

14STATUS_BUCKETS = { 

15 "proposed": "proposed", "idea": "proposed", "hypothesis-unproven": "proposed", 

16 "agreed": "to do", "pending": "to do", "open": "to do", "todo": "to do", "to do": "to do", 

17 "in-progress": "in progress", "in_progress": "in progress", "in progress": "in progress", "active": "in progress", 

18 "built": "in progress", "built-awaiting-install": "in progress", 

19 "blocked": "blocked", "on-hold": "blocked", "on hold": "blocked", "parked": "blocked", 

20 "done": "done", "completed": "done", "resolved": "done", "closed": "done", 

21 "deferred": "dropped", "dropped": "dropped", "superseded": "dropped", 

22} 

23DECISION_STATUSES = ("in force", "pending", "superseded", "retired") 

24WORK_STATUSES = ("proposed", "to do", "in progress", "blocked", "done", "dropped") 

25OPEN_GROUPS = (("d:pending", "Decisions waiting for sign-off"), ("w:blocked", "Blocked / on hold"), 

26 ("w:in progress", "In progress"), ("w:proposed", "Proposed — waiting for a yes"), 

27 ("w:to do", "Agreed, not started")) 

28KIND_ORDER = {"DECISION": 0, "WORK": 1, "THING": 2, "SESSION": 3} 

29EMPTY_MESSAGE = ("No decisions recorded yet. Agents and people add them with `niyantrak decisions write events.jsonl`; " 

30 "they appear here as a graph.")

33def bucket(raw, archived=False): 

34 """Raw ledger word → one of the six work buckets.""" 

35 r = str(raw or "").strip().lower() 

36 b = STATUS_BUCKETS.get(r) or next((v for k, v in STATUS_BUCKETS.items() if r.startswith(k)), None) 

37 return b or ("done" if archived else "to do")

40def _latest(index, key, order, e): 

41 cur = index.get(key) 

42 if cur is None or order >= cur[0]: 

43 index[key] = (order, e)

46def derive(events): 

47 links, status, retired, nodes, closed = {}, {}, {}, {}, {} 

48 for i, e in enumerate(events): 

49 order = (str(e.get("ts") or ""), i) 

50 k, src = e.get("kind"), e.get("src") 

51 if e.get("dst"): 

52 _latest(links, (k, src, e["dst"]), order, e) 

53 elif k == "STATUS" and e.get("op") == "add": 

54 _latest(status, src, order, e) 

55 if e.get("title") in ("done", "dropped") and e.get("written_by") == "reconcile": 

56 _latest(closed, src, order, e) 

57 elif k == "RETIRED": 

58 _latest(retired, src, order, e) 

59 elif k in ("DECISION", "WORK", "THING") and e.get("op") == "add": 

60 nodes.setdefault((k, src), []).append((order, e)) 

61 live = sorted(key for key, (_, e) in links.items() if e.get("op") == "add") 

62 

63 def facts(evs): 

64 evs = sorted(evs, key=lambda x: x[0]) 

65 last = evs[-1][1] 

66 pick = lambda f: next((e[f] for _, e in reversed(evs) if e.get(f)), None) 

67 return {"title": pick("title"), "text": pick("text"), "ts": max(o[0] for o, _ in evs) or None, 

68 "user": last.get("user_id"), "agent": last.get("agent_id"), "product": last.get("product"), 

69 "session": last.get("session_id")} 

70 

71 st = lambda s: bucket(status[s][1].get("title")) if s in status else None 

72 sup = {} 

73 for k, s, d in live: 

74 if k == "SUPERSEDED_BY": 

75 sup.setdefault(s, []).append(d) 

76 decisions, work, things, sessions = {}, {}, {}, {} 

77 for (k, s), evs in sorted(nodes.items(), key=lambda x: (x[0][1], x[0][0])): 

78 if k != "DECISION": 

79 continue 

80 latest = st(s) 

81 if s in sup: 

82 v = "superseded" 

83 elif (s in retired and retired[s][1].get("op") == "add") or latest == "dropped": 

84 v = "retired" 

85 elif latest in ("proposed", "to do"): 

86 v = "pending" 

87 else: 

88 v = "in force" 

89 decisions[s] = {"id": s, "kind": "DECISION", "status": v, "superseded_by": sorted(sup.get(s, [])), **facts(evs), 

90 "auto_closed": closed[s][1].get("text") if s in closed else None} 

91 for (k, s), evs in sorted(nodes.items(), key=lambda x: (x[0][1], x[0][0])): 

92 if k == "WORK" and s not in decisions: 

93 work[s] = {"id": s, "kind": "WORK", "status": st(s) or "to do", **facts(evs), 

94 "auto_closed": closed[s][1].get("text") if s in closed else None} 

95 for (k, s), evs in sorted(nodes.items(), key=lambda x: (x[0][1], x[0][0])): 

96 if k == "THING" and s not in decisions and s not in work: 

97 f = facts(evs) 

98 things[s] = {"id": s, "kind": "THING", "status": None, "title": f["title"] or s, "text": f["text"], "ts": None, 

99 "user": None, "agent": None, "product": None, "session": None, "auto_closed": None} 

100 for k, s, d in live: 

101 if k == "DECIDED_IN" and d not in decisions and d not in work and d not in things: 

102 sessions[d] = {"id": d, "kind": "SESSION", "status": None, "title": d, "text": None, "ts": None, 

103 "user": None, "agent": None, "product": None, "session": d, "auto_closed": None} 

104 ids = set(decisions) | set(work) | set(things) | set(sessions) 

105 edges = [{"source": s, "target": d, "kind": k} for k, s, d in live if s in ids and d in ids] 

106 edges.sort(key=lambda x: (x["source"], x["target"], x["kind"])) 

107 return {"decisions": decisions, "work": work, "things": things, "sessions": sessions, "edges": edges}

110def category(n): 

111 if n["kind"] == "DECISION": 

112 return "d:" + n["status"] 

113 if n["kind"] == "WORK": 

114 return "w:" + n["status"] 

115 return n["kind"]

118def all_nodes(g): 

119 ns = list(g["decisions"].values()) + list(g["work"].values()) + list(g["things"].values()) + list(g["sessions"].values()) 

120 return sorted(ns, key=lambda n: (KIND_ORDER[n["kind"]], n["id"]))

123def open_items(g): 

124 """Exact lists of what needs attention, newest first; only non-empty groups.""" 

125 out = [] 

126 ns = all_nodes(g) 

127 for key, label in OPEN_GROUPS: 

128 items = sorted((n for n in ns if category(n) == key), key=lambda n: n["id"]) 

129 items.sort(key=lambda n: str(n["ts"] or ""), reverse=True) 

130 if items: 

131 out.append({"key": key, "label": label, "kind": "decision" if key.startswith("d:") else "work", 

132 "status": key[2:], "items": [{"id": n["id"], "ts": n["ts"], "title": n["title"]} for n in items]}) 

133 return out

136def export_graph(g): 

137 """The panel's graph JSON: nodes, links, counts per status chip, open items. Deterministic for the same events.""" 

138 ns = all_nodes(g) 

139 counts = {} 

140 for n in ns: 

141 counts[category(n)] = counts.get(category(n), 0) + 1 

142 keep = ("id", "kind", "status", "title", "text", "ts", "user", "agent", "product", "session", "auto_closed") 

143 out = {"nodes": [{k: n.get(k) for k in keep} for n in ns], "links": list(g["edges"]), 

144 "counts": dict(sorted(counts.items())), "open": open_items(g), "empty": not ns} 

145 if not ns: 

146 out["message"] = EMPTY_MESSAGE 

147 return out

150def _atomic(path, data): 

151 tmp = path + ".tmp" 

152 with open(tmp, "wb") as f: 

153 f.write(data) 

154 os.replace(tmp, path)

157def materialize(store): 

158 """Current state from ALL events, saved under <root>/state/. Never the source of truth: delete it and rerun for 

159 byte-identical files. graph.json always; decisions.parquet + edges.parquet when pyarrow is installed.""" 

160 from .store import _pyarrow 

161 g = derive(store.events()) 

162 d = os.path.join(store.root, "state") 

163 os.makedirs(d, exist_ok=True) 

164 paths = [os.path.join(d, "graph.json")] 

165 _atomic(paths[0], (json.dumps(export_graph(g), sort_keys=True, indent=1, ensure_ascii=False) + "\n").encode()) 

166 pa = _pyarrow() 

167 if pa: 

168 rows = [{**n, "superseded_by": ",".join(n["superseded_by"]) or None} for n in g["decisions"].values()] 

169 cols = ("id", "status", "ts", "title", "text", "superseded_by", "user", "agent", "product", "session") 

170 for name, data, fields in (("decisions", rows, cols), ("edges", g["edges"], ("source", "target", "kind"))): 

171 table = pa.Table.from_pylist([{k: r.get(k) for k in fields} for r in data], 

172 schema=pa.schema([(k, pa.string()) for k in fields])) 

173 p = os.path.join(d, f"{name}.parquet") 

174 pa.parquet.write_table(table, p + ".tmp") 

175 os.replace(p + ".tmp", p) 

176 paths.append(p) 

177 return paths

180def duckdb_views(store): 

181 """Optional: a DuckDB connection with tables/views ev, live_edges, latest_status, decisions, work, edges that 

182 follow the same rules as derive(). Needs the duckdb package.""" 

183 import duckdb 

184 from .schema import FIELDS 

185 c = duckdb.connect() 

186 c.execute("CREATE TABLE ev (seq BIGINT, " + ", ".join(f"{f} TEXT" for f in FIELDS) + ")") 

187 rows = [(i, *[e.get(f) for f in FIELDS]) for i, e in enumerate(store.events())] 

188 if rows: 

189 c.executemany(f"INSERT INTO ev VALUES ({', '.join('?' * (len(FIELDS) + 1))})", rows) 

190 c.execute(""" 

191 CREATE VIEW live_edges AS SELECT src, dst, kind FROM ( 

192 SELECT *, row_number() OVER (PARTITION BY kind, src, dst ORDER BY coalesce(ts, '') DESC, seq DESC) AS rn 

193 FROM ev WHERE dst IS NOT NULL) WHERE rn = 1 AND op = 'add'; 

194 CREATE VIEW latest_status AS SELECT src, title AS status FROM ( 

195 SELECT *, row_number() OVER (PARTITION BY src ORDER BY coalesce(ts, '') DESC, seq DESC) AS rn 

196 FROM ev WHERE kind = 'STATUS' AND op = 'add') WHERE rn = 1; 

197 CREATE VIEW latest_retired AS SELECT src, op FROM ( 

198 SELECT *, row_number() OVER (PARTITION BY src ORDER BY coalesce(ts, '') DESC, seq DESC) AS rn 

199 FROM ev WHERE kind = 'RETIRED') WHERE rn = 1; 

200 CREATE VIEW decisions AS 

201 SELECT d.src AS id, max(d.ts) AS ts, 

202 CASE WHEN EXISTS (SELECT 1 FROM live_edges s WHERE s.kind = 'SUPERSEDED_BY' AND s.src = d.src) THEN 'superseded' 

203 WHEN EXISTS (SELECT 1 FROM latest_retired r WHERE r.src = d.src AND r.op = 'add') THEN 'retired' 

204 WHEN (SELECT status FROM latest_status l WHERE l.src = d.src) = 'dropped' THEN 'retired' 

205 WHEN (SELECT status FROM latest_status l WHERE l.src = d.src) IN ('proposed', 'to do') THEN 'pending' 

206 ELSE 'in force' END AS status 

207 FROM ev d WHERE d.kind = 'DECISION' AND d.op = 'add' GROUP BY d.src; 

208 CREATE VIEW work AS 

209 SELECT w.src AS id, max(w.ts) AS ts, coalesce(any_value(l.status), 'to do') AS status 

210 FROM ev w LEFT JOIN latest_status l ON l.src = w.src 

211 WHERE w.kind = 'WORK' AND w.op = 'add' AND w.src NOT IN (SELECT id FROM decisions) GROUP BY w.src; 

212 CREATE VIEW edges AS SELECT src, dst, kind FROM live_edges; 

213 """) 

214 return c
