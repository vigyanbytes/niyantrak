1"""Ask the 6W store (and the decision log) questions in SQL, and get the answer as a table, CSV, JSON, Markdown or Parquet. 

2 

3 niyantrak telemetry sql "SELECT who, count(*) AS n FROM ev GROUP BY 1 ORDER BY n DESC" [--format md] 

4 niyantrak telemetry ask "who ran destructive commands this week" [--as-of 2026-10-23] [--format csv] 

5 niyantrak telemetry ask --list # the questions ask understands, with their SQL 

6 

7Tables 

8 ev one row per 6W event (see docs/6W.md); which is null for events recorded before 6W; rule_ids is a LIST, entity_counts a MAP(type -> count) 

9 decisions one row per decision or work item derived from the decision log (id, kind, status, ts, title, user, 

10 agent, product); empty when there is no decision log 

11 

12Read-only by construction: the rows are copied into an in-memory DuckDB, then file and network access is switched 

13off and the configuration locked, so a query cannot read other files, write files or turn access back on. One 

14SELECT statement per call. `ask` uses no model: a question is matched by keywords to one of the built-in templates 

15below and the exact SQL is printed with the answer, so the result can be checked and re-run with `sql`. 

16Needs the optional extra: pip install niyantrak[telemetry] (pyarrow + duckdb). 

17"""

135def match(question): 

136 """The template for a question, or None: of the templates whose every keyword group appears in the question, the 

137 most specific one (most groups); ties go to the first listed.""" 

138 q = " " + re.sub(r"\s+", " ", str(question or "").lower()) + " " 

139 hits = [t for t in QUESTIONS if all(any(w in q for w in group) for group in t["keywords"])] 

140 return max(hits, key=lambda t: len(t["keywords"]), default=None)

143def window(as_of=None, days=7): 

144 """(start, end) DATE literals: the `days` days ending on as_of (default today, UTC), end exclusive.""" 

145 d = as_of if isinstance(as_of, date) else (date.fromisoformat(as_of) if as_of else datetime.now(timezone.utc).date()) 

146 return f"DATE '{d - timedelta(days=days - 1)}'", f"DATE '{d + timedelta(days=1)}'"

149def sql_for(template, as_of=None, days=7): 

150 start, end = window(as_of, days) 

151 return template["sql"].format(start=start, end=end, destructive=DESTRUCTIVE)

155def connect(store, decision_events=None): 

156 """In-memory DuckDB with tables ev and decisions, external access off and configuration locked.""" 

157 import duckdb 

158 import pyarrow as pa 

159 from . import store as telstore 

160 from ..decisions import graph 

161 con = duckdb.connect() 

162 view = store.materialize() if store is not None else None 

163 if view: 

164 con.execute("CREATE TABLE ev AS SELECT * FROM read_parquet(?)", [view]) 

165 else: 

166 con.register("ev_empty", telstore._arrow_schema().empty_table()) 

167 con.execute("CREATE TABLE ev AS SELECT * FROM ev_empty") 

168 con.unregister("ev_empty") 

169 nodes = graph.export_graph(graph.derive(decision_events))["nodes"] if decision_events else [] 

170 rows = [{k: n.get(k) for k in DECISION_COLUMNS} for n in nodes if n.get("kind") in ("DECISION", "WORK", "THING")] 

171 con.register("dec_rows", pa.Table.from_pylist(rows, schema=pa.schema([(k, pa.string()) for k in DECISION_COLUMNS]))) 

172 con.execute("CREATE TABLE decisions AS SELECT * FROM dec_rows") 

173 con.unregister("dec_rows") 

174 con.execute("SET enable_external_access = false") 

175 con.execute("SET lock_configuration = true") 

176 return con

179def run(con, sql, limit=MAX_ROWS): 

180 """(columns, rows) for one SELECT. Raises QueryError for anything else or when DuckDB refuses it.""" 

181 import duckdb 

182 try: 

183 stmts = con.extract_statements(sql) 

184 except duckdb.Error as e: 

185 raise QueryError(f"not valid SQL: {str(e).splitlines()[0]}") from None 

186 if len(stmts) != 1: 

187 raise QueryError("one statement per query") 

188 if stmts[0].type != duckdb.StatementType.SELECT: 

189 raise QueryError("read-only: only SELECT (or WITH … SELECT) queries are allowed") 

190 try: 

191 cur = con.execute(sql) 

192 cols = [d[0] for d in cur.description] 

193 rows = cur.fetchmany(limit + 1) 

194 except duckdb.Error as e: 

195 raise QueryError(str(e).splitlines()[0]) from None 

196 if len(rows) > limit: 

197 raise QueryError(f"more than {limit} rows: add a LIMIT or GROUP BY") 

198 return cols, rows

201# ── output formats ───────────────────────────────────────────────────────────────────────────────────────────── 

202def _text(v): 

203 if v is None: 

204 return "" 

205 if isinstance(v, datetime): 

206 return v.isoformat(sep=" ", timespec="seconds") 

207 if isinstance(v, (list, dict)): 

208 return json.dumps(v, default=str, sort_keys=True) 

209 return str(v)

212def _json_value(v): 

213 if isinstance(v, (datetime, date)): 

214 return v.isoformat() 

215 if isinstance(v, list): 

216 return [_json_value(x) for x in v] 

217 if isinstance(v, dict): 

218 return {str(k): _json_value(x) for k, x in v.items()} 

219 return v

222def render(cols, rows, fmt="table"): 

223 """Text output for table, csv, json or md (Parquet is written by write_parquet).""" 

224 if fmt == "csv": 

225 buf = io.StringIO() 

226 w = csv.writer(buf, lineterminator="\n") 

227 w.writerow(cols) 

228 w.writerows([[_text(v) for v in r] for r in rows]) 

229 return buf.getvalue() 

230 if fmt == "json": 

231 return json.dumps([{c: _json_value(v) for c, v in zip(cols, r)} for r in rows], indent=2, ensure_ascii=False) + "\n" 

232 cells = [[_text(v) for v in r] for r in rows] 

233 if fmt == "md": 

234 esc = lambda s: s.replace("|", "\\|") 

235 out = ["| " + " | ".join(cols) + " |", "|" + "|".join("---" for _ in cols) + "|"] 

236 out += ["| " + " | ".join(esc(c) for c in r) + " |" for r in cells] 

237 return "\n".join(out) + "\n" 

238 if fmt == "table": 

239 widths = [max([len(c)] + [len(r[i]) for r in cells]) for i, c in enumerate(cols)] 

240 line = lambda r: " ".join(v.ljust(w) for v, w in zip(r, widths)).rstrip() 

241 out = [line(cols), line(["-" * w for w in widths])] + [line(r) for r in cells] 

242 return "\n".join(out) + f"\n({len(rows)} row{'s' if len(rows) != 1 else ''})\n" 

243 raise ValueError(f"format: one of {', '.join(FORMATS)}")

246def write_parquet(con, sql, path): 

247 """Run the (already checked) query again and write the result with pyarrow — DuckDB itself cannot write files.""" 

248 import pyarrow.parquet as pq 

249 rel = con.sql(sql) 

250 table = rel.to_arrow_table() if hasattr(rel, "to_arrow_table") else rel.arrow() # duckdb 2 / 1 

251 pq.write_table(table, path, compression="zstd") 

252 return table.num_rows
