1"""Where Niyantrak keeps its state (join store, peers, panel certificate). 

2 

3NIYANTRAK_STATE_ROOT wins; otherwise /var/lib/niyantrak when running as root or when it is writable; otherwise the 

4user's own ~/.local/share/niyantrak — so the free panel works without root. On Windows (no root, no /var/lib): 

5%LOCALAPPDATA%\\Niyantrak, else the home-folder path. 

6""" 

7import os 

8 

9SYSTEM_ROOT = "/var/lib/niyantrak" 

10 

11 

12def _writable(path): 

13 p = path 

14 while p and not os.path.exists(p): 

15 p = os.path.dirname(p) 

16 return bool(p) and os.access(p, os.W_OK) 

17 

18 

19def state_root(): 

20 env = os.environ.get("NIYANTRAK_STATE_ROOT") 

21 if env: 

22 return env 

23 if os.name == "nt" or not hasattr(os, "geteuid"): 

24 local = os.environ.get("LOCALAPPDATA") 

25 if local: 

26 return os.path.join(local, "Niyantrak") 

27 elif os.geteuid() == 0 or _writable(SYSTEM_ROOT): 

28 return SYSTEM_ROOT 

29 return os.path.join(os.path.expanduser("~"), ".local", "share", "niyantrak")
