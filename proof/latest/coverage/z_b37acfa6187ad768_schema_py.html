1"""The 5W event: what an activity record may contain — and, by omission, everything it may not. 

2 

3An event says who acted, on whose behalf (whom), what kind of action, where the data went, why it was allowed or 

4stopped, and how it ended. It never holds message text, prompts, replies or identifier values: personal data shows 

5up only as per-type counts (entity_counts = {"EMAIL": 2}). validate() enforces that: 

6 * any field outside FIELDS is rejected (message, text, body, prompt, customer_id, ...) 

7 * labels are short (64 chars, 8 words; why: 96 chars, 12 words), one line; a label may be a real path 

8 (/var/log, C:\\Users\\Public, \\\\server\\share) and is checked like any other label 

9 * every string is run through the identifier recognizers (Aadhaar, PAN, IBAN, card, email, phone, ...) and 

10 rejected on a hit; long digit runs are rejected as identifier-like 

11 * session ids must already be pseudonyms (pseudonym()); the sink pseudonymises user and session ids 

12Errors name the field, never the value. 

13""" 

14import hashlib 

15import hmac 

16import re 

17import uuid 

18from datetime import datetime, timezone 

19 

20from .. import identity as _identity 

21from ..cli import detect 

22from ..recognizers import PACKS 

23 

24SCHEMA = "niyantrak.5w/1" 

25OUTCOMES = ("allowed", "masked", "local", "blocked", "needs_approval", "approved", "rejected", "error") 

26SOURCES = ("json", "otlp") 

27REQUIRED = ("ts", "who", "what", "where", "why", "outcome") 

28LABELS = ("who", "whom", "what", "where", "why", "agent", "route", "machine") 

29FIELDS = frozenset(LABELS + ("ts", "outcome", "rule_ids", "entity_counts", "session", "source", "event_id", "schema", 

30 "who_id")) 

31COLUMNS = ("event_id", "ts", "who", "who_id", "whom", "what", "where", "why", "agent", "route", "machine", "session", 

32 "rule_ids", "outcome", "entity_counts", "source", "schema") 

33 

34MAX_LABEL, MAX_WORDS = 64, 8 

35MAX_WHY, MAX_WHY_WORDS = 96, 12 

36MAX_RULES, MAX_ENTITY_TYPES, MAX_COUNT = 16, 32, 1_000_000 

37LABEL_RX = re.compile(r"^[\w][\w .:/()+#&,'·-]*$") 

38# A label may also be a real path: POSIX absolute (/var/log), Windows drive (C:\Users, D:/data) or UNC 

39# (\\server\share). Path characters add only @ = ~ and the backslash; shell punctuation (; | ` $ < > " *) stays out, 

40# and the length, word, one-line, recognizer and digit-run guards below apply to paths exactly as to any label. 

41_PATH_CHARS = r"[\w .:/\\()+#&,'·@=~-]*" 

42PATH_RX = re.compile(r"^(?:/|[A-Za-z]:[\\/]|\\\\\w)" + _PATH_CHARS + "$") 

43RULE_RX = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._:-]{0,63}$") 

44ENTITY_RX = re.compile(r"^[A-Z][A-Z0-9_]{1,31}$") 

45PSEUDONYM_RX = re.compile(r"^p-[0-9a-f]{12}$") 

46EVENT_ID_RX = re.compile(r"^[0-9a-f]{32}$") 

47DIGIT_RUN = re.compile(r"\d{9,}") 

48ALL_PACKS = list(PACKS)

51class EventError(ValueError): 

52 """An event that cannot be stored. The message names the field, never its value.""" 

53

55def pseudonym(value, salt): 

56 """Stable, salted, one-way stand-in for a user or session id: p-<12 hex>.""" 

57 return "p-" + hmac.new(str(salt).encode(), str(value).encode(), hashlib.sha256).hexdigest()[:12]

60def _no_identifiers(field, s): 

61 if PSEUDONYM_RX.match(s): 

62 return 

63 if DIGIT_RUN.search(s.replace(" ", "").replace("-", "")): 

64 raise EventError(f"{field}: looks like an identifier value") 

65 if detect(s, ALL_PACKS): 

66 raise EventError(f"{field}: contains personal data (only per-type counts may be recorded)") 

67

69def _label(field, v, required=False): 

70 if v is None or v == "": 

71 if required: 

72 raise EventError(f"{field}: required") 

73 return None 

74 if not isinstance(v, str): 

75 raise EventError(f"{field}: must be text") 

76 v = v.strip() 

77 limit, words = (MAX_WHY, MAX_WHY_WORDS) if field == "why" else (MAX_LABEL, MAX_WORDS) 

78 if len(v) > limit or len(v.split()) > words or "\n" in v or "\r" in v: 

79 raise EventError(f"{field}: free text is not allowed (at most {limit} characters, {words} words, one line)") 

80 if not (LABEL_RX.match(v) or PATH_RX.match(v)): 

81 raise EventError(f"{field}: unexpected characters") 

82 _no_identifiers(field, v) 

83 return v

86def _ts(v): 

87 if isinstance(v, bool) or v is None: 

88 raise EventError("ts: required (ISO 8601 or epoch seconds)") 

89 try: 

90 if isinstance(v, (int, float)): 

91 d = datetime.fromtimestamp(float(v), tz=timezone.utc) 

92 else: 

93 s = str(v).strip() 

94 d = datetime.fromisoformat(s[:-1] + "+00:00" if s.endswith("Z") else s) 

95 d = d.replace(tzinfo=timezone.utc) if d.tzinfo is None else d.astimezone(timezone.utc) 

96 except (ValueError, OverflowError, OSError): 

97 raise EventError("ts: not a timestamp") from None 

98 return d.strftime("%Y-%m-%dT%H:%M:%S.%fZ") 

99

101def parse_ts(s): 

102 """ISO string from validate() -> naive UTC datetime (how the store keeps it).""" 

103 return datetime.strptime(s, "%Y-%m-%dT%H:%M:%S.%fZ")

106def validate(event): 

107 """Return a normalised copy of a 5W event, or raise EventError.""" 

108 if not isinstance(event, dict): 

109 raise EventError("event: must be an object") 

110 extra = sorted(str(k) for k in event if k not in FIELDS) 

111 if extra: 

112 raise EventError(f"field not allowed: {', '.join(k[:40] for k in extra[:5])} (5W events carry no values or text)") 

113 for f in REQUIRED: 

114 if event.get(f) in (None, ""): 

115 raise EventError(f"{f}: required") 

116 out = {"schema": SCHEMA, "ts": _ts(event["ts"])} 

117 if event.get("schema") not in (None, SCHEMA): 

118 raise EventError("schema: unsupported version") 

119 eid = event.get("event_id") 

120 if eid is not None and not (isinstance(eid, str) and EVENT_ID_RX.match(eid)): 

121 raise EventError("event_id: 32 lowercase hex characters") 

122 out["event_id"] = eid or uuid.uuid4().hex 

123 for f in LABELS: 

124 out[f] = _label(f, event.get(f), required=f in REQUIRED) 

125 if event["outcome"] not in OUTCOMES: 

126 raise EventError(f"outcome: one of {', '.join(OUTCOMES)}") 

127 out["outcome"] = event["outcome"] 

128 sess = event.get("session") 

129 if sess is not None and not (isinstance(sess, str) and PSEUDONYM_RX.match(sess)): 

130 raise EventError("session: must be a pseudonym (p-<12 hex>), never the raw id") 

131 out["session"] = sess 

132 who_id = event.get("who_id") # the acting person's directory id (niyantrak.identity), never a raw identifier 

133 if who_id is not None and not _identity.valid_stable_id(who_id): 

134 raise EventError("who_id: a directory id like sid:S-1-5-21-..., kanidm:uid:70001 or local:uid:1000@<machine>") 

135 out["who_id"] = who_id 

136 src = event.get("source") 

137 if src is not None and src not in SOURCES: 

138 raise EventError(f"source: one of {', '.join(SOURCES)}") 

139 out["source"] = src 

140 rules = event.get("rule_ids") or [] 

141 if not isinstance(rules, list) or len(rules) > MAX_RULES: 

142 raise EventError(f"rule_ids: a list of at most {MAX_RULES} rule ids") 

143 for r in rules: 

144 if not (isinstance(r, str) and RULE_RX.match(r)): 

145 raise EventError("rule_ids: rule ids are short codes like RB-ROUTE-001") 

146 _no_identifiers("rule_ids", r) 

147 out["rule_ids"] = list(rules) 

148 counts = event.get("entity_counts") or {} 

149 if not isinstance(counts, dict) or len(counts) > MAX_ENTITY_TYPES: 

150 raise EventError(f"entity_counts: an object of at most {MAX_ENTITY_TYPES} types") 

151 for k, n in counts.items(): 

152 if not (isinstance(k, str) and ENTITY_RX.match(k)): 

153 raise EventError("entity_counts: keys are entity types like EMAIL or AADHAAR") 

154 if isinstance(n, bool) or not isinstance(n, int) or not 0 <= n <= MAX_COUNT: 

155 raise EventError("entity_counts: counts are whole numbers >= 0") 

156 out["entity_counts"] = dict(counts) 

157 return out
