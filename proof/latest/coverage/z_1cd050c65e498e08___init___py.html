1"""As-built documents from facts: node cards and the join store → Markdown + HTML pages and an SVG network map. 

2 

3 pages = render(collect(join.load(store), local={"card": card.collect(), "checks": [...]})) 

4 write("<state root>/docs", pages) # returns the files that changed; same input → byte-identical files 

5 

6Output: index.md/.html (fleet overview), node-<node id>.md/.html (hardware, OS, addresses, roles, governance 

7checks), network-map.svg, manifest.json. No timestamps of generation, every list sorted, so a regenerate with 

8the same facts rewrites nothing. Card text comes from other machines and is untrusted: it is escaped in HTML, 

9SVG and Markdown. SSH host keys and token hashes are never written (keys as a sha256 only). 

10"""

37def _node(card, **extra): 

38 card = card if isinstance(card, dict) else {} 

39 osd = card.get("os") if isinstance(card.get("os"), dict) else {} 

40 addr = card.get("addresses") if isinstance(card.get("addresses"), dict) else {} 

41 gpus = [g for g in card.get("gpu") or [] if isinstance(g, dict)] 

42 return {"node_id": str(card.get("node_id") or ""), "hostname": str(card.get("hostname") or card.get("node_id") or "?"), 

43 "role": card.get("role"), "os": " ".join(str(x) for x in (osd.get("id"), osd.get("version")) if x) or None, 

44 "kernel": osd.get("kernel"), "arch": osd.get("arch"), "cores": (card.get("cpu") or {}).get("cores"), 

45 "memory_gb": card.get("memory_gb"), 

46 "gpu": sorted(f"{g.get('name', '?')}" + (f" ({round(g['vram_mb'] / 1024)} GB)" if g.get("vram_mb") else "") for g in gpus), 

47 "ipv4": sorted(str(a) for a in addr.get("ipv4") or []), "ipv6": sorted(str(a) for a in addr.get("ipv6") or []), 

48 "card_sha256": hashlib.sha256(json.dumps(card, sort_keys=True, separators=(",", ":")).encode()).hexdigest(), 

49 "host_key_sha256": None, "checks": [], "decided_by": None, "decided_at": None, "requested_at": None, **extra}

52def collect(join_store, local=None): 

53 """Nodes for the documents: this machine first (when given), then every machine in the join store by node id.""" 

54 nodes, seen = [], set() 

55 if local and isinstance(local.get("card"), dict): 

56 n = _node(local["card"], state="this machine", source="this machine", 

57 checks=[{"name": str(c.get("name")), "ok": bool(c.get("ok"))} for c in local.get("checks") or []]) 

58 nodes.append(n) 

59 seen.add(n["node_id"]) 

60 joined = (join_store or {}).get("nodes") or {} 

61 for nid in sorted(joined): 

62 rec = joined[nid] 

63 if nid in seen or not re.fullmatch(r"nyk-[0-9a-f]{1,64}", str(nid)): 

64 continue 

65 pub = " ".join(str(rec.get("host_pubkey") or "").split()[:2]) 

66 n = _node(rec.get("card"), state=str(rec.get("state") or "pending"), source="joined", 

67 decided_by=rec.get("decided_by"), decided_at=rec.get("decided_at"), requested_at=rec.get("requested_at"), 

68 host_key_sha256=hashlib.sha256(pub.encode()).hexdigest() if pub else None) 

69 n["node_id"] = nid # the store's key is authoritative 

70 if rec.get("card_sha256"): 

71 n["card_sha256"] = str(rec["card_sha256"]) 

72 nodes.append(n) 

73 return {"nodes": nodes}

77def _md_cell(v): 

78 s = "—" if v in (None, "", []) else str(v) 

79 s = html.escape(s, quote=False).replace("|", "\\|") 

80 return re.sub(r"\s+", " ", s)

87def to_md(title, blocks): 

88 out = [f"# {_md_cell(title)}", ""] 

89 for b in blocks: 

90 if b[0] == "h": 

91 out += [f"## {b[1]}", ""] 

92 elif b[0] == "p": 

93 out += [_md_cell(b[1]), ""] 

94 elif b[0] == "img": 

95 out += [f"![{b[1]}]({b[2]})", ""] 

96 elif b[0] == "table": 

97 head, rows = b[1], b[2] 

98 if not rows: 

99 out += ["_none recorded_", ""] 

100 continue 

101 cell = lambda c: f"[{_md_cell(c[1])}]({c[2]})" if isinstance(c, tuple) else _md_cell(c) 

102 out += ["| " + " | ".join(head) + " |", "|" + "|".join("---" for _ in head) + "|"] 

103 out += ["| " + " | ".join(cell(c) for c in r) + " |" for r in rows] + [""] 

104 return "\n".join(out).rstrip("\n") + "\n"

107def to_html(title, blocks, svg=None, back=True): 

108 e = lambda v: html.escape("—" if v in (None, "", []) else str(v)) 

109 out = ["<!doctype html>", '<html lang="en">', "<head>", '<meta charset="utf-8">', 

110 '<meta name="viewport" content="width=device-width,initial-scale=1">', f"<title>{e(title)}</title>", 

111 f"<style>\n{CSS}</style>", "</head>", "<body>", "<main>"] 

112 if back: 

113 out.append('<p><a href="index.html">← Fleet as built</a></p>') 

114 out.append(f"<h1>{e(title)}</h1>") 

115 for b in blocks: 

116 if b[0] == "h": 

117 out.append(f"<h2>{e(b[1])}</h2>") 

118 elif b[0] == "p": 

119 out.append(f"<p>{e(b[1])}</p>") 

120 elif b[0] == "img": 

121 out.append(svg if svg else f'<p><a href="{e(b[2])}">{e(b[1])}</a></p>') 

122 elif b[0] == "table": 

123 if not b[2]: 

124 out.append('<p class="muted">none recorded</p>') 

125 continue 

126 cell = lambda c: f'<a href="{e(c[2].replace(".md", ".html"))}">{e(c[1])}</a>' if isinstance(c, tuple) else e(c) 

127 out.append('<div class="wrap"><table><thead><tr>' + "".join(f"<th>{e(h)}</th>" for h in b[1]) + "</tr></thead><tbody>") 

128 out += ["<tr>" + "".join(f"<td>{cell(c)}</td>" for c in r) + "</tr>" for r in b[2]] 

129 out.append("</tbody></table></div>") 

130 out += ["</main>", "</body>", "</html>"] 

131 return "\n".join(out) + "\n"

146def fleet_blocks(nodes): 

147 if not nodes: 

148 return [("p", INTRO), ("p", "No machines recorded yet. Approve machines under Fleet in the panel, then regenerate.")] 

149 counts = {} 

150 for n in nodes: 

151 counts[n["state"]] = counts.get(n["state"], 0) + 1 

152 summary = f"{len(nodes)} machine(s): " + ", ".join(f"{v} {k}" for k, v in sorted(counts.items())) + "." 

153 machines = [[n["hostname"], n["node_id"], n["state"], n["role"], n["os"], n["cores"], _gb(n["memory_gb"]), 

154 ", ".join(n["gpu"]), ", ".join(n["ipv4"] + n["ipv6"]), _md_link("details", _page_name(n) + ".md")] for n in nodes] 

155 gov = [[n["hostname"], n["state"], n["decided_by"], n["decided_at"], _checks_passing(n)] for n in nodes] 

156 return [("p", INTRO), ("p", summary), 

157 ("h", "Machines"), ("table", ["host", "node id", "state", "role", "os", "cpu cores", "memory", "gpu", "addresses", "page"], machines), 

158 ("h", "Network map"), ("img", "Network map", "network-map.svg"), 

159 ("h", "Governance"), ("table", ["host", "state", "decided by", "decided at", "checks passing"], gov)]

162def node_blocks(n): 

163 hw = [["cpu cores", n["cores"]], ["memory", _gb(n["memory_gb"])], ["gpu", ", ".join(n["gpu"]) or "none"], ["architecture", n["arch"]]] 

164 osr = [["operating system", n["os"]], ["kernel", n["kernel"]]] 

165 addrs = [["IPv4", a] for a in n["ipv4"]] + [["IPv6", a] for a in n["ipv6"]] 

166 gov = [["join state", n["state"]]] 

167 if n["source"] == "joined": 

168 gov += [["card signature verified at join", "yes (the join store records only verified cards)"], 

169 ["requested at", n["requested_at"]], ["decided by", n["decided_by"]], ["decided at", n["decided_at"]], 

170 ["host key sha256", n["host_key_sha256"]]] 

171 gov += [["card sha256", n["card_sha256"]]] 

172 gov += [[c["name"], "PASS" if c["ok"] else "FAIL"] for c in n["checks"]] 

173 return [("p", f"Node {n['node_id']} · {n['source']}."), 

174 ("h", "Hardware"), ("table", ["fact", "value"], hw), 

175 ("h", "Operating system"), ("table", ["fact", "value"], osr), 

176 ("h", "Addresses"), ("table", ["family", "address"], addrs), 

177 ("h", "Roles"), ("table", ["role", "source"], [[n["role"], "node card"]] if n["role"] else []), 

178 ("h", "Governance checks"), ("table", ["check", "result"], gov)]

181# ── network map ───────────────────────────────────────────────────────────────────────────────────────── 

182def network_of(n): 

183 for a in n["ipv4"]: 

184 try: 

185 return str(ipaddress.ip_network(f"{a}/24", strict=False)) 

186 except ValueError: 

187 continue 

188 for a in n["ipv6"]: 

189 try: 

190 ip = ipaddress.ip_address(a.split("%", 1)[0]) 

191 except ValueError: 

192 continue 

193 if not ip.is_link_local: 

194 return str(ipaddress.ip_network(f"{ip}/64", strict=False)) 

195 return "IPv6 link-local only" if n["ipv6"] else "no address"

198def network_svg(nodes): 

199 groups = {} 

200 for n in nodes: 

201 groups.setdefault(network_of(n), []).append(n) 

202 names = sorted(groups) 

203 col, gap, pad, row = 220, 24, 20, 56 

204 e = lambda s: html.escape(str(s)) 

205 width = max(pad * 2 + len(names) * col + max(len(names) - 1, 0) * gap, 360) 

206 tallest = max((len(v) for v in groups.values()), default=0) 

207 box_h = 44 + tallest * row 

208 height = 70 + box_h + 60 

209 out = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{width}" height="{height}" viewBox="0 0 {width} {height}" ' 

210 'role="img" aria-label="Network map" font-family="Inter, system-ui, sans-serif" font-size="12">', 

211 f'<rect width="{width}" height="{height}" fill="#ffffff"/>', 

212 f'<text x="{pad}" y="24" font-size="14" font-weight="700" fill="#1c1814">Network map</text>'] 

213 if not names: 

214 out.append(f'<text x="{pad}" y="60" fill="#6b6357">No machines recorded yet.</text>') 

215 else: 

216 out.append(f'<line x1="{pad + col // 2}" y1="44" x2="{pad + (len(names) - 1) * (col + gap) + col // 2}" y2="44" ' 

217 'stroke="#6b6357" stroke-width="2"/>') 

218 for i, name in enumerate(names): 

219 x = pad + i * (col + gap) 

220 out += [f'<line x1="{x + col // 2}" y1="44" x2="{x + col // 2}" y2="60" stroke="#6b6357" stroke-width="2"/>', 

221 f'<rect x="{x}" y="60" width="{col}" height="{box_h}" rx="8" fill="#f7f2e7" stroke="#d6cfc2"/>', 

222 f'<text x="{x + 10}" y="80" font-weight="600" fill="#1c1814">{e(name)}</text>'] 

223 for j, n in enumerate(sorted(groups[name], key=lambda n: (n["hostname"], n["node_id"]))): 

224 y = 92 + j * row 

225 colour = STATE_COLOURS.get(n["state"], "#57534e") 

226 addr = (n["ipv4"] + [a for a in n["ipv6"] if not a.startswith("fe80")] + n["ipv6"] + ["—"])[0] 

227 out += [f'<rect x="{x + 10}" y="{y}" width="{col - 20}" height="44" rx="6" fill="#ffffff" stroke="{colour}" stroke-width="2"/>', 

228 f'<text x="{x + 20}" y="{y + 18}" font-weight="600" fill="#1c1814">{e(n["hostname"][:28])}</text>', 

229 f'<text x="{x + 20}" y="{y + 34}" fill="#3a332b">{e(addr[:30])} · {e(n["state"])}</text>'] 

230 ly = 70 + box_h + 30 

231 for k, (state, colour) in enumerate(STATE_COLOURS.items()): 

232 lx = pad + k * 110 

233 out += [f'<rect x="{lx}" y="{ly - 10}" width="12" height="12" rx="3" fill="{colour}"/>', 

234 f'<text x="{lx + 18}" y="{ly}" fill="#3a332b">{e(state)}</text>'] 

235 out.append("</svg>") 

236 return "\n".join(out) + "\n"

240def leaks(text): 

241 return [p.pattern[:30] for p in SECRET_PATTERNS if p.search(text)] + (["home path"] if re.search(r"/home/[a-z]", text) else [])

244def render(inputs): 

245 nodes = inputs.get("nodes") or [] 

246 svg = network_svg(nodes) 

247 fb = fleet_blocks(nodes) 

248 pages = {"index.md": to_md(TITLE, fb), "index.html": to_html(TITLE, fb, svg=svg, back=False), "network-map.svg": svg} 

249 listing = [{"file": "index.html", "md": "index.md", "title": TITLE, "kind": "fleet"}, 

250 {"file": "network-map.svg", "md": None, "title": "Network map", "kind": "map"}] 

251 for n in sorted(nodes, key=lambda n: n["node_id"]): 

252 name, blocks = _page_name(n), node_blocks(n) 

253 pages[name + ".md"] = to_md(n["hostname"], blocks) 

254 pages[name + ".html"] = to_html(n["hostname"], blocks) 

255 listing.append({"file": name + ".html", "md": name + ".md", "title": n["hostname"], "kind": "node"}) 

256 files = sorted(list(pages) + ["manifest.json"]) 

257 pages["manifest.json"] = json.dumps({"generator": "niyantrak docs", "version": 1, "pages": listing, "files": files}, 

258 indent=1, sort_keys=True) + "\n" 

259 found = sorted({x for body in pages.values() for x in leaks(body)}) 

260 if found: 

261 raise ValueError(f"refusing to write documents: possible secret or personal path ({', '.join(found)})") 

262 return dict(sorted(pages.items()))

265def write(out_dir, pages): 

266 """Write changed files only (atomic), remove files a previous run generated that are gone now. Returns changed.""" 

267 os.makedirs(out_dir, exist_ok=True) 

268 old = set() 

269 try: 

270 with open(os.path.join(out_dir, "manifest.json")) as f: 

271 old = set(json.load(f).get("files") or []) 

272 except (OSError, ValueError): 

273 pass 

274 changed = [] 

275 for name, body in pages.items(): 

276 path, data = os.path.join(out_dir, name), body.encode() 

277 try: 

278 with open(path, "rb") as f: 

279 if f.read() == data: 

280 continue 

281 except OSError: 

282 pass 

283 with open(path + ".tmp", "wb") as f: 

284 f.write(data) 

285 os.replace(path + ".tmp", path) 

286 changed.append(name) 

287 for name in sorted(old - set(pages)): 

288 if "/" not in name and os.path.isfile(os.path.join(out_dir, name)): 

289 os.unlink(os.path.join(out_dir, name)) 

290 return sorted(changed)
