21def _outcome(case): 

22 if case.find("failure") is not None: 

23 return "failed" 

24 if case.find("error") is not None: 

25 return "error" 

26 if case.find("skipped") is not None: 

27 return "skipped"

31def _split_classname(classname): 

32 """tests.unit.test_beta.TestGroup -> ("tests/unit/test_beta.py", "TestGroup").""" 

33 parts = classname.split(".") 

34 for i, p in enumerate(parts): 

35 if p.startswith("test_") or p == "test": 

36 return "/".join(parts[:i + 1]) + ".py", "::".join(parts[i + 1:]) 

37 return classname or "(unknown)", ""

40def _count(counts, outcome): 

41 counts["total"] += 1 

42 if outcome == "error": 

43 counts["errors"] += 1 

44 counts["failed"] += 1 

45 else: 

46 counts[outcome] += 1

49def parse_junit(path): 

50 """pytest JUnit XML -> {"tests": counts, "time": seconds, "files": [per-file counts + tests by name]}. 

51 Errors count as failed (and are also reported under "errors").""" 

52 out = {"tests": _empty_counts(), "time": 0.0, "files": []} 

53 try: 

54 root = ET.parse(path).getroot() 

55 except (OSError, ET.ParseError): 

56 return out 

57 suites = [root] if root.tag == "testsuite" else root.findall("testsuite") 

58 files = {} 

59 for s in suites: 

60 out["time"] += float(s.get("time") or 0) 

61 for case in s.iter("testcase"): 

62 fname, prefix = _split_classname(case.get("classname", "")) 

63 name = f"{prefix}::{case.get('name', '')}" if prefix else case.get("name", "") 

64 oc, t = _outcome(case), float(case.get("time") or 0) 

65 f = files.setdefault(fname, {"file": fname, **_empty_counts(), "time": 0.0, "tests": []}) 

66 _count(f, oc) 

67 _count(out["tests"], oc) 

68 f["time"] = round(f["time"] + t, 3) 

69 f["tests"].append({"name": name, "outcome": oc, "time": round(t, 3)}) 

70 out["time"] = round(out["time"], 3) 

71 out["files"] = sorted(files.values(), key=lambda f: f["file"]) 

72 return out

75def parse_coverage(path): 

76 """coverage.py JSON report -> {"total": pct, "files": {path: pct}} (one decimal).""" 

77 try: 

78 d = json.load(open(path, encoding="utf-8")) 

79 return {"total": round(d["totals"]["percent_covered"], 1), 

80 "files": {k: round(v["summary"]["percent_covered"], 1) for k, v in sorted(d.get("files", {}).items())}} 

81 except (OSError, ValueError, KeyError, TypeError): 

82 return {"total": None, "files": {}}

85def leak_findings(output): 

86 """Number of findings from tools/leak-scan.py output, or None when it did not report.""" 

87 m = re.search(r"leak-scan: \w+ \((\d+) finding", output or "")

91def scrub(text, root): 

92 """Drop the machine's host name (JUnit hostname=) and the absolute checkout path from kept evidence.""" 

93 text = re.sub(r'\s+hostname="[^"]*"', "", text) 

94 if root: 

95 text = text.replace(root.rstrip("/") + "/", "").replace(root.rstrip("/"), ".") 

96 return text

99def build_summary(junit, coverage, leak_findings, git, timings, date): 

100 t = dict(junit["tests"]) 

101 leak = "not run" if leak_findings is None else "clean" if leak_findings == 0 else f"{leak_findings} findings"

105 "timings": dict(timings), "files": junit["files"], "coverage_files": coverage.get("files", {}), "screens": []}

123def read_history(path, limit=None): 

124 out = [] 

125 try: 

126 lines = open(path, encoding="utf-8").read().splitlines() 

127 except OSError: 

128 return out 

129 for line in lines: 

130 try: 

131 v = json.loads(line) 

132 except ValueError: 

133 continue 

134 if isinstance(v, dict): 

135 out.append(v) 

136 return out[-limit:] if limit else out

148def safe_path(base, rel): 

149 """Absolute path of an existing regular file strictly inside base, or None. Refuses .., absolute paths, NUL and 

150 symlinks that lead outside.""" 

151 if not rel or "\x00" in rel or rel.startswith(("/", "\\")) or os.path.isabs(rel): 

152 return None 

153 root = os.path.realpath(base) 

154 full = os.path.realpath(os.path.join(root, rel)) 

155 if not full.startswith(root + os.sep) or not os.path.isfile(full): 

156 return None 

157 return full

193def _trend(values, top, color, unit=""): 

194 vals = [v for v in values if isinstance(v, (int, float))] 

195 if not vals: 

196 return '<p class="muted">No runs yet.</p>' 

197 top = top or max(vals) or 1 

198 w, h, pad = 300, 120, 8 

199 n = len(vals) 

200 xs = [pad + (w - 2 * pad) * (i / (n - 1) if n > 1 else 0.5) for i in range(n)] 

201 ys = [h - pad - (h - 2 * pad) * min(v, top) / top for v in vals] 

202 pts = " ".join(f"{x:.1f},{y:.1f}" for x, y in zip(xs, ys)) 

203 dots = "".join(f'<circle cx="{x:.1f}" cy="{y:.1f}" r="2.5" fill="var({color})"/>' for x, y in zip(xs, ys)) 

204 return (f'<svg viewBox="0 0 {w} {h}" role="img" aria-label="trend, latest {vals[-1]}{unit}">' 

205 f'<line class="g" x1="{pad}" x2="{w - pad}" y1="{pad}" y2="{pad}"/><line class="g" x1="{pad}" x2="{w - pad}" y1="{h - pad}" y2="{h - pad}"/>' 

206 f'<text x="{pad}" y="{pad + 10}">{top:g}{unit}</text>' 

207 f'<polyline fill="none" stroke="var({color})" stroke-width="2" points="{pts}"/>{dots}</svg>')
