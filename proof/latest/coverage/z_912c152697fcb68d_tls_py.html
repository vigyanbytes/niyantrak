1"""TLS for the join listener: a self-signed certificate made on first start, pinned by its SHA-256 fingerprint. 

2 

3No certificate authority is involved: the operator reads the fingerprint in the panel (it is part of the join 

4one-liner), and the joining machine refuses any server whose certificate does not hash to it — before the token is 

5sent. The certificate is made with the `openssl` command; the private key is written with mode 0600 in a 0700 folder. 

6""" 

7import hashlib 

8import http.client 

9import os 

10import re 

11import shutil 

12import ssl 

13import subprocess 

14 

15CERT, KEY = "join-cert.pem", "join-key.pem" 

16 

17 

18def ensure_cert(state_dir, common_name="niyantrak-join", days=3650): 

19 """Return (cert_path, key_path), creating a self-signed EC P-256 certificate if there is none yet.""" 

20 os.makedirs(state_dir, mode=0o700, exist_ok=True) 

21 os.chmod(state_dir, 0o700) 

22 cert, key = os.path.join(state_dir, CERT), os.path.join(state_dir, KEY) 

23 if not (os.path.isfile(cert) and os.path.isfile(key)): 

24 if not shutil.which("openssl"): 

25 raise RuntimeError("the join listener needs the openssl command to create its certificate") 

26 subprocess.run(["openssl", "req", "-x509", "-newkey", "ec", "-pkeyopt", "ec_paramgen_curve:prime256v1", 

27 "-nodes", "-keyout", key, "-out", cert, "-days", str(days), "-subj", f"/CN={common_name}"], 

28 check=True, capture_output=True, preexec_fn=lambda: os.umask(0o077)) 

29 os.chmod(key, 0o600) 

30 return cert, key

33def fingerprint(cert_path): 

34 """Lower-case hex SHA-256 of the certificate (DER), the value the client pins.""" 

35 with open(cert_path) as f: 

36 return hashlib.sha256(ssl.PEM_cert_to_DER_cert(f.read())).hexdigest() 

37 

38 

39def normalise_pin(pin): 

40 """Accept 'sha256:<hex>', plain hex, or colon-separated hex; return 64 lower-case hex characters.""" 

41 h = str(pin or "").strip().lower() 

42 h = re.sub(r"[:\s]", "", h[7:] if h.startswith("sha256:") else h) 

43 if not re.fullmatch(r"[0-9a-f]{64}", h): 

44 raise ValueError("certificate pin must be a SHA-256 fingerprint (64 hex characters)") 

45 return h

48def server_context(cert, key): 

49 ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER) 

50 ctx.minimum_version = ssl.TLSVersion.TLSv1_2 

51 ctx.load_cert_chain(cert, key) 

52 return ctx

59class PinnedHTTPSConnection(http.client.HTTPSConnection): 

60 """HTTPS that trusts exactly one certificate: the one whose SHA-256 equals `pin` (checked before any request).""" 

61 

62 def __init__(self, host, port=None, pin=None, timeout=30): 

63 ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT) 

64 ctx.minimum_version = ssl.TLSVersion.TLSv1_2 

65 ctx.check_hostname = False # self-signed: identity comes from the pin, not from a CA or a name 

66 ctx.verify_mode = ssl.CERT_NONE 

67 super().__init__(host, port, timeout=timeout, context=ctx) 

68 self.pin = normalise_pin(pin) 

69 

70 def connect(self): 

71 super().connect() 

72 got = hashlib.sha256(self.sock.getpeercert(binary_form=True) or b"").hexdigest() 

73 if got != self.pin: 

74 self.sock.close() 

75 raise PinMismatch(f"server certificate sha256:{got} does not match the pinned certificate")
