1"""Install / uninstall the gate hook into the CLI configs under a GIVEN home directory. 

2 

3 claude <home>/.claude/settings.json hooks.PreToolUse (matcher "*") 

4 codex <home>/.codex/hooks.json hooks.PreToolUse (matcher ".*") 

5 gemini <home>/.gemini/settings.json hooks.BeforeTool (matcher ".*", timeout in ms) 

6 antigravity <home>/.gemini/antigravity-cli/hooks.json group "niyantrak" {enabled, PreToolUse} 

7 opencode <home>/.config/opencode/plugin/niyantrak-gate.js (tool.execute.before plugin) 

8 

9Rules: merge, never clobber -- every other key and every other hook is kept in order; our entry is recognised by 

10MARK and replaced in place, so installing twice changes nothing. A file that is not valid JSON is left untouched and 

11reported. Before the first change to an existing file a one-time ``<file>.niyantrak-backup`` is written. Uninstall 

12removes only our entries (and containers that become empty, and a file that becomes ``{}``). 

13 

14The home is always explicit: CLAUDE_CONFIG_DIR, CODEX_HOME and XDG_CONFIG_HOME are deliberately ignored here so an 

15install aimed at one home can never land in another. 

16""" 

17import json 

18import os 

19import shlex 

20import subprocess 

21import sys 

22from dataclasses import dataclass 

23 

24CLIS = ("claude", "codex", "gemini", "antigravity", "opencode") 

25MARK = "gate check --cli" 

26GROUP = "niyantrak" 

27 

28JSON_HOOKS = { # cli -> (relative path, event key, matcher, timeout) 

29 "claude": (".claude/settings.json", "PreToolUse", "*", 10), 

30 "codex": (".codex/hooks.json", "PreToolUse", ".*", 10), 

31 "gemini": (".gemini/settings.json", "BeforeTool", ".*", 10000), 

32} 

33# Codex runs a user hook only after the operator has reviewed and trusted it ("N hooks need review before they can 

34# run"; hooks/list reports it as untrusted until then). Niyantrak never writes that trust for the operator. 

35INSTALL_NOTES = { 

36 "codex": "; Codex runs it only after you review and trust it in Codex (new hooks need review before they run)", 

37} 

38AGY_PATH = ".gemini/antigravity-cli/hooks.json" 

39OPENCODE_PATH = ".config/opencode/plugin/niyantrak-gate.js" 

40 

41OPENCODE_PLUGIN = """\ 

42// Niyantrak gate for OpenCode. Managed by `niyantrak gate install --cli opencode`; 

43// remove with `niyantrak gate uninstall --cli opencode`. Marker: gate check --cli opencode 

44// 

45// OpenCode loads plugin/*.js from its config dir and calls tool.execute.before before every tool call; 

46// throwing blocks the call and the message goes back to the model. The decision is made by the same 

47// Niyantrak gate every other CLI uses. If the gate cannot answer, the call is blocked (fail closed). 

48import { spawnSync } from "node:child_process"; 

49 

50const GATE = __ARGV__; 

51 

52export const NiyantrakGate = async () => ({ 

53 "tool.execute.before": async (input, output) => { 

54 const payload = JSON.stringify({ tool: input && input.tool, args: (output && output.args) || {} }); 

55 const r = spawnSync(GATE[0], GATE.slice(1), { input: payload, encoding: "utf8", timeout: 15000 }); 

56 let res = null; 

57 try { res = JSON.parse(r.stdout || ""); } catch (e) { res = null; } 

58 if (!res || (res.decision !== "allow" && res.decision !== "deny")) { 

59 throw new Error("Niyantrak gate unavailable (" + (r.error ? r.error.message : "exit " + r.status) + 

60 "); tool call blocked. Remove the gate with: niyantrak gate uninstall --cli opencode"); 

61 } 

62 if (res.decision === "deny") throw new Error(res.reason || "Blocked by Niyantrak gate"); 

63 }, 

64}); 

65"""

71 path: str 

72 ok: bool

77def default_command(): 

78 return [sys.executable, "-m", "niyantrak.cli"]

81def hook_argv(cli, command=None): 

82 return list(command or default_command()) + ["gate", "check", "--cli", cli]

85def hook_command(cli, command=None): 

86 argv = hook_argv(cli, command) 

87 return subprocess.list2cmdline(argv) if os.name == "nt" else shlex.join(argv)

90def target_path(cli, home): 

91 rel = JSON_HOOKS[cli][0] if cli in JSON_HOOKS else AGY_PATH if cli == "antigravity" else OPENCODE_PATH 

92 return os.path.join(os.fspath(home), *rel.split("/"))

95def _ours(obj, cli=None): 

96 s = json.dumps(obj) 

97 return MARK in s and (cli is None or f"{MARK} {cli}" in s) 

98

100def _read_json(path): 

101 """(doc, text) -- ({}, None) when absent; raises ValueError when not a JSON object.""" 

102 if not os.path.isfile(path): 

103 return {}, None 

104 with open(path, encoding="utf-8") as fh: 

105 text = fh.read() 

106 doc = json.loads(text) if text.strip() else {} 

107 if not isinstance(doc, dict): 

108 raise ValueError("not a JSON object") 

109 return doc, text 

110 

111 

112def _write(path, text, old_text): 

113 if old_text is not None: 

114 backup = path + ".niyantrak-backup" 

115 if not os.path.exists(backup): 

116 with open(backup, "w", encoding="utf-8") as fh: 

117 fh.write(old_text) 

118 os.makedirs(os.path.dirname(path), exist_ok=True) 

119 tmp = path + ".niyantrak-tmp" 

120 with open(tmp, "w", encoding="utf-8") as fh: 

121 fh.write(text) 

122 if os.path.exists(path): 

123 os.chmod(tmp, os.stat(path).st_mode & 0o7777) 

124 os.replace(tmp, path) 

125 

126 

127def _dump(doc): 

128 return json.dumps(doc, indent=2) + "\n" 

129 

130 

131def _entry(cli, command): 

132 _rel, _event, matcher, timeout = JSON_HOOKS.get(cli, (None, None, ".*", 10)) 

133 return {"matcher": matcher, "hooks": [{"type": "command", "command": hook_command(cli, command), 

134 "timeout": timeout}]}

136 

137def _install_json(cli, path, doc, command): 

138 event = JSON_HOOKS[cli][1] 

139 hooks = doc.setdefault("hooks", {}) 

140 if not isinstance(hooks, dict) or not isinstance(hooks.setdefault(event, []), list): 

141 raise ValueError(f'"hooks.{event}" has an unexpected shape') 

142 new, out, placed = _entry(cli, command), [], False 

143 for e in hooks[event]: # replace our entry where it was; drop duplicates 

144 if not _ours(e): 

145 out.append(e) 

146 elif not placed: 

147 out.append(new) 

148 placed = True 

149 if not placed: 

150 out.append(new) 

151 hooks[event] = out 

152 return doc 

153 

154 

155def _install_agy(cli, path, doc, command): 

156 doc[GROUP] = {"enabled": True, "PreToolUse": [_entry(cli, command)]} 

157 return doc 

158 

159 

160def _uninstall_json(cli, doc): 

161 event = JSON_HOOKS[cli][1] 

162 hooks = doc.get("hooks") 

163 if not isinstance(hooks, dict) or not isinstance(hooks.get(event), list): 

164 return doc 

165 hooks[event] = [e for e in hooks[event] if not _ours(e)] 

166 if not hooks[event]: 

167 del hooks[event] 

168 if not hooks: 

169 del doc["hooks"] 

170 return doc

173def install(cli, home, command=None): 

174 path = target_path(cli, home) 

175 if cli == "opencode": 

176 text = OPENCODE_PLUGIN.replace("__ARGV__", json.dumps(hook_argv(cli, command))) 

177 old = open(path, encoding="utf-8").read() if os.path.isfile(path) else None 

178 if old is not None and MARK not in old: 

179 return Result(cli, path, False, False, f"{path} exists and is not ours; left untouched") 

180 if old == text: 

181 return Result(cli, path, True, False, "already current") 

182 _write(path, text, None) 

183 return Result(cli, path, True, True, "installed") 

184 try: 

185 doc, old = _read_json(path) 

186 doc = (_install_agy if cli == "antigravity" else _install_json)(cli, path, doc, command) 

187 except ValueError as exc: 

188 return Result(cli, path, False, False, f"{path}: not valid JSON or unexpected shape ({exc}); left untouched") 

189 text = _dump(doc) 

190 if text == old: 

191 return Result(cli, path, True, False, "already current") 

192 _write(path, text, old) 

193 return Result(cli, path, True, True, "installed" + INSTALL_NOTES.get(cli, ""))

196def uninstall(cli, home): 

197 path = target_path(cli, home) 

198 if not os.path.isfile(path): 

199 return Result(cli, path, True, False, "not installed") 

200 if cli == "opencode": 

201 if MARK not in open(path, encoding="utf-8").read(): 

202 return Result(cli, path, False, False, f"{path} is not ours; left untouched") 

203 os.remove(path) 

204 return Result(cli, path, True, True, "removed") 

205 try: 

206 doc, old = _read_json(path) 

207 except ValueError as exc: 

208 return Result(cli, path, False, False, f"{path}: not valid JSON ({exc}); left untouched") 

209 if cli == "antigravity": 

210 doc.pop(GROUP, None) 

211 else: 

212 doc = _uninstall_json(cli, doc) 

213 if not doc: 

214 os.remove(path) 

215 return Result(cli, path, True, True, "removed (file was only the gate)") 

216 text = _dump(doc) 

217 if text == old or json.loads(old or "{}") == doc: 

218 return Result(cli, path, True, False, "not installed") 

219 _write(path, text, None) 

220 return Result(cli, path, True, True, "removed")

223def is_installed(cli, home): 

224 path = target_path(cli, home) 

225 try: 

226 with open(path, encoding="utf-8") as fh: 

227 text = fh.read() 

228 except OSError: 

229 return False 

230 if cli == "opencode": 

231 return f"{MARK} opencode" in text 

232 try: 

233 doc = json.loads(text) 

234 except ValueError: 

235 return False 

236 if cli == "antigravity": 

237 return isinstance(doc, dict) and _ours(doc.get(GROUP), cli) 

238 hooks = doc.get("hooks") if isinstance(doc, dict) else None 

239 return isinstance(hooks, dict) and any(_ours(e, cli) for e in hooks.get(JSON_HOOKS[cli][1]) or [])
